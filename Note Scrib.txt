function doGet() {
  return ContentService.createTextOutput("Aktif").setMimeType(ContentService.MimeType.TEXT);
}

function doPost(e) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName("Sheet1") || ss.getActiveSheet();
    const d = JSON.parse(e.postData.contents);

    // --- CEK APAKAH NISN SUDAH ADA ---
    const dataSemua = sheet.getDataRange().getValues();
    const kolomNISN = 2; // Kolom C = indeks ke-2 (mulai dari 0)
    let sudahAda = false;

    for (let i = 1; i < dataSemua.length; i++) { // mulai dari baris 2 (lewati judul)
      if (String(dataSemua[i][kolomNISN]).trim() === String(d.nisn).trim()) {
        sudahAda = true;
        break;
      }
    }

    if (sudahAda) {
      return ContentService.createTextOutput(JSON.stringify({
        status: "ganda",
        pesan: "NISN ini sudah terdaftar sebelumnya, tidak bisa mendaftar lagi."
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // --- JIKA BELUM ADA, BARU SIMPAN ---
    const nisgcText = "'" + d.nisgc;
    const nisnText = "'" + d.nisn;
    const hpText = "'" + d.hp;

    sheet.appendRow([
      new Date(),
      d.nama,
      nisnText,
      d.tempat_lahir,
      d.tgl_lahir,
      d.alamat,
      hpText,
      d.program,
      nisgcText
    ]);

    return ContentService.createTextOutput(JSON.stringify({
      status: "sukses",
      pesan: "Data berhasil disimpan"
    })).setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error",
      pesan: err.message
    })).setMimeType(ContentService.MimeType.JSON);
  }
}
